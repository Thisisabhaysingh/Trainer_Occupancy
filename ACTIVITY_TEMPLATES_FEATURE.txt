### Next Steps (Backend):

Create these backend routes in `routes/activities.js`:

```javascript
// Get templates for manager's department
router.get('/templates', auth, checkRole(['manager']), async (req, res) => {
  try {
    const templates = await Activity.findAll({
      where: { 
        departmentId: req.user.departmentId,
        isTemplate: true 
      }
    });
    res.json(templates);
  } catch (error) {
    res.status(500).json({ error: 'Server error' });
  }
});

// Create template
router.post('/templates', auth, checkRole(['manager']), async (req, res) => {
  try {
    const { title } = req.body;
    const template = await Activity.create({
      title,
      departmentId: req.user.departmentId,
      isTemplate: true
    });
    res.status(201).json({ message: 'Template created', template });
  } catch (error) {
    res.status(500).json({ error: 'Server error' });
  }
});

// Update template
router.put('/templates/:id', auth, checkRole(['manager']), async (req, res) => {
  try {
    const { title } = req.body;
    const template = await Activity.findOne({
      where: { 
        id: req.params.id,
        departmentId: req.user.departmentId,
        isTemplate: true
      }
    });
    if (!template) return res.status(404).json({ error: 'Template not found' });
    
    template.title = title;
    await template.save();
    res.json({ message: 'Template updated', template });
  } catch (error) {
    res.status(500).json({ error: 'Server error' });
  }
});

// Delete template
router.delete('/templates/:id', auth, checkRole(['manager']), async (req, res) => {
  try {
    const template = await Activity.findOne({
      where: { 
        id: req.params.id,
        departmentId: req.user.departmentId,
        isTemplate: true
      }
    });
    if (!template) return res.status(404).json({ error: 'Template not found' });
    
    await template.destroy();
    res.json({ message: 'Template deleted' });
  } catch (error) {
    res.status(500).json({ error: 'Server error' });
  }
});
```
